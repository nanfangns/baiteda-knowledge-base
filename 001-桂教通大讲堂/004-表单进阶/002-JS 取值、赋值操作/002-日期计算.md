## 功能说明

1. 选择开通日期
2. 填写时长
3. 自动生成到期时间

## 代码示例


```javascript
const ctx = exports.ctx;
const utils = exports.utils;
const http = utils.getHttp()
const isMobile = utils.getDevice() === 'mobile'
const moment = exports.env.moment

function calcStopDate() {
  const startDate = ctx.getState('LewZdGi5WE')
  const applyTerm = ctx.getState('hOo6BggDNv')

  ctx.setState('HXBbpMijhQ', moment(Number(startDate)).add(applyTerm, 'month').valueOf())
}

export function ktrqChange (payload) {
  calcStopDate()
}
export function sqsjChange (payload) {
  calcStopDate()
}
```
