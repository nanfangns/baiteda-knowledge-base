### 1.1.2非form类型的组件如何校验？

通过监听平台提交前事件，可以阻止表单提交


```typescript
import { onEngineEvent } from './onEngineEvent'

onEngineEvent('engine-submit', async (payload: EventPayload) => {
  return new Promise((reslove, reject) => {
    setTimeout(() => {
      reslove(false)
    }, 1000)
  })
})
```


​

通过一个switch来决定 页面是否可以提交，完整示例如下：


```typescript
<template>
  <a-switch v-model:checked="checked" />
</template>

<script setup lang="ts">
import { computed, inject, Ref, unref, ref } from 'vue'
import Property from '../property'
import {
  useInstanceProps,
  PAGE_STATUS,
} from '@byteluck-fe/custom-control-sdk/runtime'
import { Engine } from '@byteluck-fe/model-driven-engine/dist/types';

const checked = ref(false)

const props = defineProps<{
  instance: typeof Property.Runtime
  rowIndex?: number
  name: string | string[]
  value: string[]
}>()
import { onEngineEvent } from './onEngineEvent'
const buttonProps = useInstanceProps(props.instance)
const isDisabled = computed(
  () =>
    props.instance.pageStatus === PAGE_STATUS.PRINT ||
    buttonProps.value.defaultState === 'readonly'
)


onEngineEvent('engine-submit', async () => {
  return new Promise((reslove, reject) => {
    setTimeout(() => {
      reslove(checked.value)
    }, 1000)
  })
})
</script>
```
