# 2.请求对象


```plain
/**
 * @description: 请求对象基类
 * @author: croyson
 * @create: 2019-12-05 21:51
 **/
public class BaseDto {

    /**
     * 租户ID
     */
    private String tenantId;

    /**
     * 应用ID
     */
    private String appId;

    private String userId;
```


```plain
public class SelectOneDto extends BaseDto {

    /**
     * 服务code
     */
    private String svcCode;

    /**
     * 条件 key:主键字段名称，value:值
     */
    private Map<String, Object> saveDatas = new HashMap<>();
```


```plain
public class SelectMoreDto extends BaseDto {

    /**
     * 服务code
     */
    private String svcCode;

    /**
     * 条件
     */
    private DataSvcPageQuery query;
```
