### 查询类型 枚举

1. EQ(0, "等于查询"),
2. GT(1, "大于查询"),
3. LT(2, "小于查询"),
4. IN(3, "包含查询"),
5. LIKE(4, "模糊查询"),
6. RANGE(5, "范围查询"),
7. GE(6, "大于等于"),
8. LE(7, "小于等于"),
9. CONTAINS\_ONE(8, "包含任一"),
10. CONCAT\_LIKE(9, "多字段拼接模糊查询"),
11. CONTAINS\_IN\_WITH\_SUB\_DEPT(10, "含子部门"),
12. NOT\_EQ(11, "不等于"),
13. CONCAT\_NOT\_EQ(12, "拼接不等于"),
14. IS\_NULL(13, "为空"),
15. IS\_NOT\_NULL(14, "不为空"),
16. EQ\_FIELD(15, "等于某个字段"),//如：查询树的根节点的时候需要通过 id = parent\_id 的方式查询
17. IS\_NULL\_OR\_EQ\_FIELD(16, "为空或等于某个字段"), //如：查询树的根节点的时候需要通过 id = parent\_id 的方式查询
18. NOT\_EQ\_FIELD(17, "不等于某个字段"), //如：查询树的子节点的时候需要通过 parent\_id = 'value' and id != parent\_id 的方式查询
